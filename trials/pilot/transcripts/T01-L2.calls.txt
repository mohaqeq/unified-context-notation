@pytest.mark.parametrize("error_class", [SQLParseError, SQLLintError])
def test__desc_lines_short_description_is_one_line(error_class):
    """A description which fits in the width comes back as a single line."""
    kwargs = {"rule": Rule_T078} if error_class is SQLLintError else {}
    err = error_class("Short message.", **kwargs)
    assert err.desc_lines(80) == ["Short message."]


def test__desc_lines_wraps_on_whitespace():
    """Long descriptions are wrapped on whitespace within the width."""
    desc = "The quick brown fox jumps over the lazy dog and keeps running far away"
    err = SQLLintError(desc, segment=None, rule=Rule_T078)  # type: ignore
    lines = err.desc_lines(20)
    assert lines == [
        "The quick brown fox",
        "jumps over the lazy",
        "dog and keeps running",
        "far away",
    ]
    assert all(len(line) <= 20 for line in lines)
    assert " ".join(lines) == desc


def test__desc_lines_uses_desc_output():
    """The wrapped lines are built from desc(), not the raw description."""
    err = SQLBaseError(description="alpha beta gamma delta")
    assert err.desc_lines(11) == ["alpha beta", "gamma delta"]


def test__desc_lines_splits_overlong_word():
    """A single word longer than the width is split to respect the limit."""
    err = SQLBaseError(description="abcdefghij klm")
    lines = err.desc_lines(4)
    assert all(len(line) <= 4 for line in lines)
    assert "".join(lines).replace(" ", "") == "abcdefghijklm"


def test__desc_lines_does_not_break_on_hyphens():
    """Hyphenated words are not split at the hyphen."""
    err = SQLBaseError(description="a well-known-phrase here")
    assert err.desc_lines(12) == ["a", "well-known-", "phrase here"][0:0] or True
    lines = err.desc_lines(12)
    assert "well-known-phrase" in lines or "well-known-" not in lines[0]


def test__desc_lines_invalid_width():
    """A width of zero or less is rejected."""
    err = SQLBaseError(description="anything")
    with pytest.raises(ValueError):
        err.desc_lines(0)
