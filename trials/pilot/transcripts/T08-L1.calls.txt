@pytest.mark.parametrize(
    "sql, expected",
    [
        ("SELECT 1 -- noqa: LT05\n", ["-- noqa: LT05"]),
        ("SELECT 1 -- noqa\n", ["-- noqa"]),
        ("SELECT 1 # noqa: LT05\n", ["# noqa: LT05"]),
        ("SELECT 1 /* noqa: LT05 */\n", ["/* noqa: LT05 */"]),
        ("SELECT 1 /*noqa*/\n", ["/*noqa*/"]),
        # Chained "--" sections: the last one counts.
        ("SELECT 1 -- note -- noqa: LT05\n", ["-- note -- noqa: LT05"]),
        ("SELECT 1 -- noqa: LT05 -- note\n", []),
        # Mentions of noqa do not count, and matching is case-sensitive.
        ("SELECT 1 -- see the noqa docs\n", []),
        ("SELECT 1 -- NOQA\n", []),
        ("SELECT 1 /* a note about noqa */\n", []),
        # Plain comments and no comments at all.
        ("SELECT 1 -- a plain comment\n", []),
        ("SELECT 1\n", []),
    ],
)
def test__parser__base_segments_get_noqa_comments(sql, expected):
    """Test BaseSegment.get_noqa_comments() on the comment forms it recognises."""
    tree = Linter(dialect="ansi").parse_string(sql).tree
    assert [seg.raw for seg in tree.get_noqa_comments()] == expected


def test__parser__base_segments_get_noqa_comments_source_order():
    """Test that get_noqa_comments() returns matches in source order."""
    sql = "SELECT\n    a, -- noqa: LT05\n    b /* noqa */\nFROM t -- noqa: AL01\n"
    tree = Linter(dialect="ansi").parse_string(sql).tree
    results = tree.get_noqa_comments()
    assert [seg.raw for seg in results] == [
        "-- noqa: LT05",
        "/* noqa */",
        "-- noqa: AL01",
    ]
    positions = [seg.pos_marker.source_slice.start for seg in results]
    assert positions == sorted(positions)


def test__parser__base_segments_get_noqa_comments_includes_self():
    """Test that get_noqa_comments() includes the segment itself if it is a comment."""
    tree = Linter(dialect="ansi").parse_string("SELECT 1 -- noqa\n").tree
    noqa_comment = next(tree.recursive_crawl("comment"))
    assert noqa_comment.get_noqa_comments() == [noqa_comment]

    tree = Linter(dialect="ansi").parse_string("SELECT 1 -- a note\n").tree
    plain_comment = next(tree.recursive_crawl("comment"))
    assert plain_comment.get_noqa_comments() == []
EOF
