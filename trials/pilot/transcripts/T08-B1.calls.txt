def test__parser__base_segments_get_noqa_comments_source_order():
    """Test .get_noqa_comments() returns noqa comments in source order."""
    sql = (
        "select 1 -- noqa: LT05\n"
        "from t /* noqa */\n"
        "-- see the noqa docs\n"
        "# noqa\n"
        "select 2 -- note -- noqa: CP01\n"
    )
    tree = Linter(dialect="ansi").parse_string(sql).tree
    found = tree.get_noqa_comments()
    assert [seg.raw for seg in found] == [
        "-- noqa: LT05",
        "/* noqa */",
        "# noqa",
        "-- note -- noqa: CP01",
    ]
    assert all(seg.is_type("inline_comment", "block_comment") for seg in found)


@pytest.mark.parametrize(
    "comment,expected",
    [
        ("-- noqa", True),
        ("--noqa", True),
        ("-- noqa: LT05", True),
        ("-- note -- noqa: LT05", True),
        ("# noqa: LT05", True),
        ("/* noqa */", True),
        ("/*noqa: LT05*/", True),
        ("/* note -- noqa */", True),
        ("-- see the noqa docs", False),
        ("-- NOQA", False),
        ("-- noqa -- note", False),
        ("-- plain comment", False),
        ("/* plain comment */", False),
    ],
)
def test__parser__base_segments_get_noqa_comments_directive(comment, expected):
    """Test which comments count as a noqa directive."""
    tree = Linter(dialect="ansi").parse_string(f"select 1 {comment}\n").tree
    assert bool(tree.get_noqa_comments()) is expected


def test__parser__base_segments_get_noqa_comments_includes_self():
    """Test .get_noqa_comments() includes the segment itself if it is a comment."""
    tree = Linter(dialect="ansi").parse_string("select 1 -- noqa\n").tree
    comment = tree.get_noqa_comments()[0]
    assert comment.get_noqa_comments() == [comment]
