@pytest.mark.parametrize(
    "description,width,expected",
    [
        # A short description comes back as a single line.
        ("Short", 80, ["Short"]),
        # Lines are broken on whitespace and never exceed the width.
        (
            "The quick brown fox jumps over the lazy dog",
            10,
            ["The quick", "brown fox", "jumps over", "the lazy", "dog"],
        ),
        # A word longer than the width is split so no line exceeds it.
        ("abcdefghijklmno", 5, ["abcde", "fghij", "klmno"]),
    ],
)
def test__error_desc_lines(description, width, expected):
    """Wrap the description of an error into lines of at most ``width``."""
    assert SQLParseError(description).desc_lines(width) == expected


def test__error_desc_lines_uses_desc():
    """desc_lines wraps the same text that desc returns."""
    template = TemplatedFile.from_string("foobar")
    segment = RawSegment("foobar", PositionMarker(slice(0, 6), slice(0, 6), template))
    err = SQLLintError(
        "A long description which needs to be wrapped",
        segment=segment,
        rule=Rule_T078,
    )
    lines = err.desc_lines(12)
    assert lines == ["A long", "description", "which needs", "to be", "wrapped"]
    assert all(len(line) <= 12 for line in lines)
    assert " ".join(lines) == err.desc()


def test__error_desc_lines_invalid_width():
    """A non-positive width is rejected."""
    with pytest.raises(ValueError):
        SQLBaseError("Foo").desc_lines(0)
