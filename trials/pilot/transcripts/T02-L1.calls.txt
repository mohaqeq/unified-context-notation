"""Information API."""

import platform
import sys
from importlib import metadata

from sqlfluff.core import Linter, dialect_readout
from sqlfluff.core.dialects import DialectTuple
from sqlfluff.core.linter import RuleTuple


def list_rules() -> list[RuleTuple]:
    """Return a list of available rule tuples."""
    linter = Linter()
    return linter.rule_tuples()


def list_dialects() -> list[DialectTuple]:
    """Return a list of available dialect info."""
    return list(dialect_readout())


def version_info() -> dict[str, str]:
    """Return the versions of sqlfluff and Python in use.

    This is intended for bug reports, so that the exact sqlfluff and
    Python build can be identified.

    Returns:
        A dictionary with the string keys ``"sqlfluff"`` (the installed
        sqlfluff version, as in ``sqlfluff.__version__``), ``"python"``
        (the interpreter version as ``major.minor.micro``) and
        ``"implementation"`` (the lowercase Python implementation name,
        e.g. ``"cpython"`` or ``"pypy"``).
    """
    version = sys.version_info
    return {
        # Read the metadata directly rather than importing
        # ``sqlfluff.__version__``, which is defined after this package
        # is imported and so would be a circular import.
        "sqlfluff": metadata.version("sqlfluff"),
        "python": f"{version.major}.{version.minor}.{version.micro}",
        "implementation": platform.python_implementation().lower(),
    }
EOF
