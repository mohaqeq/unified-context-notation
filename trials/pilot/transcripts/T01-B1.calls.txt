def test__desc_lines_short_description_is_one_line():
    """A description that fits within the width comes back as one line."""
    err = SQLLintError(description="Short message.")
    assert err.desc_lines(80) == ["Short message."]


@pytest.mark.parametrize(
    "description, width, expected",
    [
        (
            "The quick brown fox jumps over the lazy dog",
            15,
            ["The quick brown", "fox jumps over", "the lazy dog"],
        ),
        # Exactly at the limit stays on one line.
        ("abcd efgh", 9, ["abcd efgh"]),
        # Whitespace is collapsed and used as the break point.
        ("one   two\nthree", 7, ["one two", "three"]),
    ],
)
def test__desc_lines_wraps_on_whitespace(description, width, expected):
    """Descriptions wrap on whitespace, with no line over the width."""
    err = SQLLintError(description=description)
    lines = err.desc_lines(width)
    assert lines == expected
    assert all(len(line) <= width for line in lines)


def test__desc_lines_splits_overlong_word():
    """A single word longer than the width is split so lines stay within it."""
    err = SQLLintError(description="abcdefghijklmnop")
    lines = err.desc_lines(5)
    assert all(len(line) <= 5 for line in lines)
    assert "".join(lines) == "abcdefghijklmnop"


def test__desc_lines_uses_desc_on_base_error():
    """Subclasses without a description fall back to desc() via the base class."""
    err = SQLBaseError(description="Some base error text here")
    assert err.desc_lines(10) == ["Some base", "error text", "here"]


def test__desc_lines_rejects_non_positive_width():
    """A width below 1 is rejected."""
    err = SQLLintError(description="Anything")
    with pytest.raises(ValueError):
        err.desc_lines(0)
