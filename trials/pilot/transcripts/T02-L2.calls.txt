def version_info() -> dict[str, str]:
    """Return the sqlfluff and Python versions in use.

    Intended for bug reports, so the reporter can say which sqlfluff and
    Python build they were running.

    Returns:
        A dict with the string entries ``"sqlfluff"`` (the installed sqlfluff
        version), ``"python"`` (the interpreter version as
        ``major.minor.micro``) and ``"implementation"`` (the lowercase Python
        implementation name, e.g. ``"cpython"`` or ``"pypy"``).
    """
    # Imported here rather than at module level: the top-level ``sqlfluff``
    # package imports this module (via ``sqlfluff.api``) before it defines
    # ``__version__``, so a module-level import would be circular.
    import sqlfluff

    version = sys.version_info
    return {
        "sqlfluff": sqlfluff.__version__,
        "python": f"{version.major}.{version.minor}.{version.micro}",
        "implementation": platform.python_implementation().lower(),
    }
EOF

def test__api__info_version_info():
    """The version info dict reports sqlfluff, Python and implementation."""
    info = version_info()
    assert set(info.keys()) == {"sqlfluff", "python", "implementation"}
    assert all(isinstance(value, str) for value in info.values())
    assert info["sqlfluff"] == sqlfluff.__version__
    assert info["python"] == "{}.{}.{}".format(*sys.version_info[:3])
    assert info["implementation"] == platform.python_implementation().lower()
    assert info["implementation"] == info["implementation"].lower()
EOF
