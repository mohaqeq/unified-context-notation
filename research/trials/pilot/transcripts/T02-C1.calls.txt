def test__api__info_version_info():
    """Check the version info dict reports sqlfluff and Python versions."""
    info = version_info()
    assert set(info.keys()) == {"sqlfluff", "python", "implementation"}
    assert all(isinstance(value, str) for value in info.values())
    # The sqlfluff entry matches the package version.
    assert info["sqlfluff"] == sqlfluff.__version__
    # The python entry is major.minor.micro of the running interpreter.
    assert info["python"] == (
        f"{sys.version_info.major}.{sys.version_info.minor}."
        f"{sys.version_info.micro}"
    )
    assert re.fullmatch(r"\d+\.\d+\.\d+", info["python"])
    # The implementation is the lowercased interpreter implementation name.
    assert info["implementation"] == platform.python_implementation().lower()
    assert info["implementation"] == info["implementation"].lower()
EOF
