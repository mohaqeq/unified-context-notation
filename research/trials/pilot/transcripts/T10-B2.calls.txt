def test__api__format_violations_layout():
    """Violations are formatted like the CLI, one per line, without a newline."""
    violations = [
        {
            "code": "CP01",
            "description": "Keywords must be consistently upper case.",
            "name": "capitalisation.keywords",
            "start_line_no": 1,
            "start_line_pos": 1,
            "warning": False,
        },
        {
            "code": "PRS",
            "description": "Unparsable section.",
            "name": "",
            "start_line_no": 12,
            "start_line_pos": 105,
            "warning": False,
        },
    ]
    assert sqlfluff.api.format_violations(violations) == (
        "L:   1 | P:   1 | CP01 | Keywords must be consistently upper case. "
        "[capitalisation.keywords]\n"
        "L:  12 | P: 105 |  PRS | Unparsable section."
    )


def test__api__format_violations_long_description_not_wrapped():
    """Long descriptions are kept on a single line."""
    description = " ".join(["word"] * 60)
    violations = [
        {
            "code": "LT01",
            "description": description,
            "name": "layout.spacing",
            "start_line_no": 3,
            "start_line_pos": 2,
            "warning": False,
        }
    ]
    result = sqlfluff.api.format_violations(violations)
    assert "\n" not in result
    assert result == (
        f"L:   3 | P:   2 | LT01 | {description} [layout.spacing]"
    )
