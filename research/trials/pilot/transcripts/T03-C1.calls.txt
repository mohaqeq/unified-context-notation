        # Equal references overlap.
        (("schema", "tbl"), ("schema", "tbl"), True),
        (("tbl",), ("tbl",), True),
        # A trailing part of the other reference overlaps.
        (("tbl",), ("schema", "tbl"), True),

        (("a", "c"), ("a", "b", "c"), False),
        # Comparison is case-sensitive and exact.
        (("Tbl",), ("tbl",), False),
        (("SCHEMA", "tbl"), ("schema", "tbl"), False),
        (("tb",), ("tbl",), False),
        # Empty references never overlap anything.
        ((), (), False),
        ((), ("tbl",), False),
        (("tbl",), (), False),
    ],
)
def test__parser__helper_references_overlap(first, second, expected):
    """Test references_overlap, in both argument orders."""
    assert references_overlap(first, second) is expected
    assert references_overlap(second, first) is expected
