def test__api__format_violations_empty():
    """An empty list of violations formats to an empty string."""
    assert sqlfluff.format_violations([]) == ""


def test__api__format_violations():
    """Violations are formatted one per line in the CLI layout."""
    expected = (
        "L:   1 | P:   1 | CP01 | Keywords must be consistently upper case."
        " [capitalisation.keywords]\n"
        "L:   1 | P:   1 | LT09 | Select targets should be on a new line unless "

        "L:   1 | P:   7 | LT01 | Expected only single space before star '*'. "
        "Found '  '. [layout.spacing]\n"
        "L:   1 | P:   9 | AM04 | Query produces an unknown number of result "
        "columns. [ambiguous.column_count]\n"
        "L:   1 | P:  12 | AL03 | Column expression without alias. Use explicit "
        "`AS` clause. [aliasing.expression]\n"
        "L:   1 | P:  20 | CP01 | Keywords must be consistently upper case."
        " [capitalisation.keywords]"
    )
    result = sqlfluff.format_violations(lint_result)
    assert result == expected.rstrip("\n").replace("\n", "\n")
    assert not result.endswith("\n")
    assert "\x1b[" not in result


def test__api__format_violations_no_name_and_multidigit():
    """Missing or empty names add no suffix; wide numbers are not truncated."""
    violations = [
        {
            "code": "PRS",
            "start_line_no": 123,
            "start_line_pos": 4567,
            "description": "Unparsable section.",
            "name": "",
        },
        {
            "code": "LT01",
            "start_line_no": 2,
            "start_line_pos": 3,
            "description": "Spacing.",
        },
    ]
    assert sqlfluff.format_violations(violations) == (
        "L: 123 | P:4567 |  PRS | Unparsable section.\n"
        "L:   2 | P:   3 | LT01 | Spacing."
    )
