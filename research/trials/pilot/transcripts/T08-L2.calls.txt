@pytest.mark.parametrize(
    "comment,is_noqa",
    [
        ("-- noqa", True),
        ("--noqa: LT05", True),
        ("# noqa: LT05", True),
        ("/* noqa */", True),
        ("/*noqa*/", True),
        ("-- note -- noqa: LT05", True),
        ("-- see the noqa docs", False),
        ("-- Noqa", False),
        ("-- nonoqa", False),
        ("/* a noqa mention */", False),
        ("-- noqa -- note", False),
    ],
)
def test__parser__base_segments_get_noqa_comments_directive(comment, is_noqa):
    """Test which comment forms count as noqa directives."""
    tree = Linter(dialect="ansi").parse_string(f"SELECT 1 {comment}\nFROM t\n").tree
    assert [seg.raw for seg in tree.get_noqa_comments()] == (
        [comment] if is_noqa else []
    )


def test__parser__base_segments_get_noqa_comments_order_and_subtree():
    """Test noqa comments are returned in source order and scoped to the subtree."""
    sql = (
        "-- noqa: LT01\n"
        "SELECT a -- noqa: AL01\n"
        "FROM t /* noqa */\n"
        "-- see the noqa docs\n"
    )
    tree = Linter(dialect="ansi").parse_string(sql).tree
    found = tree.get_noqa_comments()
    assert [seg.raw for seg in found] == [
        "-- noqa: LT01",
        "-- noqa: AL01",
        "/* noqa */",
    ]
    select = next(tree.recursive_crawl("select_statement"))
    raws = [seg.raw for seg in select.get_noqa_comments()]
    assert "-- noqa: LT01" not in raws
    assert "-- noqa: AL01" in raws
    # A noqa comment segment is its own result.
    assert found[0].get_noqa_comments() == [found[0]]
EOF

def _is_noqa_comment(raw: str) -> bool:
    text = raw.strip()
    if text.startswith("/*"):
        text = text[2:]
    if text.endswith("*/"):
        text = text[:-2]
    if text.startswith("#"):
        text = text[1:]
    return text.split("--")[-1].strip().startswith("noqa")

cases = ["-- noqa", "--noqa: LT05", "# noqa: LT05", "/* noqa */", "/*noqa*/",
         "-- note -- noqa: LT05", "-- see the noqa docs", "-- Noqa", "-- nonoqa",
         "/* a noqa mention */", "-- noqa -- note", " noqa */"]
for c in cases:
    print(repr(c), _is_noqa_comment(c))
