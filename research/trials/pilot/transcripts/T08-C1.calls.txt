def test__parser__base_segments_get_noqa_comments():
    """Test .get_noqa_comments() returns only noqa directive comments, in order."""
    sql = (
        "-- noqa: LT05\n"
        "SELECT 1 -- see the noqa docs\n"
        "FROM t /* noqa: disable=all */\n"
        "WHERE a = 1 -- note -- noqa: CP01\n"
        "# noqa\n"
        "/* NOQA */\n"
    )
    tree = Linter(dialect="ansi").parse_string(sql).tree
    assert [seg.raw for seg in tree.get_noqa_comments()] == [
        "-- noqa: LT05",
        "/* noqa: disable=all */",
        "-- note -- noqa: CP01",
        "# noqa",
    ]


def test__parser__base_segments_get_noqa_comments_self_and_subtree():
    """Test .get_noqa_comments() on a comment itself and on a non-comment."""
    tree = Linter(dialect="ansi").parse_string("SELECT 1 -- noqa\n").tree
    comments = tree.get_noqa_comments()
    assert len(comments) == 1
    # A noqa comment includes itself when called on it.
    assert comments[0].get_noqa_comments() == comments
    # A segment without any comments below it returns an empty list.
    select_kw = next(tree.recursive_crawl("keyword"))
    assert select_kw.get_noqa_comments() == []


def test__parser__base_segments_get_noqa_comments_none():
    """Test .get_noqa_comments() on a file with no noqa directives."""
    sql = "SELECT 1 -- just a note\n/* a block */\n"
    tree = Linter(dialect="ansi").parse_string(sql).tree
    assert tree.get_noqa_comments() == []
